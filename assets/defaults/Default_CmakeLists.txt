cmake_minimum_required(VERSION 3.16)

project({ProjectName})

set(CMAKE_CXX_STANDARD 20)

set(ENGINE_PATH {EnginePath})

add_subdirectory(${ENGINE_PATH} ${ENGINE_PATH}/cmake-build-debug)

file(GLOB_RECURSE SOURCE_FILES CONFIGURE_DEPENDS Source/*.cpp Source/*.h)

set(IMGUI_PATH ${ENGINE_PATH}/extern/imgui)

file(GLOB LIBS CONFIGURE_DEPENDS
        ${ENGINE_PATH}/lib/glad/src/glad.c
)

add_library(${CMAKE_PROJECT_NAME} SHARED ${SOURCE_FILES} ${LIBS})

target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE Engine)

target_include_directories(${CMAKE_PROJECT_NAME} PUBLIC Source)

message(STATUS "CMake is including the following source files for project:")
foreach(file ${SOURCE_FILES})
    message(STATUS "  ${file}")
endforeach()

message(STATUS "CMake is including the following library files:")
foreach(file ${LIBS})
    message(STATUS "  ${file}")
endforeach()
